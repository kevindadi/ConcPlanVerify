```rust
use std::sync::mpsc::sync_channel;
use std::thread;

fn s1(ch_tx: std::sync::mpsc::SyncSender<i32>) {
    ch_tx.send(1).expect("receiver should be alive");
}

fn r(ch_rx: std::sync::mpsc::Receiver<i32>) {
    let _ = ch_rx.recv().expect("sender should be alive");
}

fn main() {
    let (ch_tx, ch_rx) = sync_channel::<i32>(0);

    let sender = thread::spawn(move || s1(ch_tx));
    let receiver = thread::spawn(move || r(ch_rx));

    sender.join().expect("sender panicked");
    receiver.join().expect("receiver panicked");

    println!("DONE done=1");
}
```
