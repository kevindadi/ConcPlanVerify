use std::sync::{Arc, Mutex};
use std::thread;

fn a(
    a: Arc<Mutex<bool>>,
    b: Arc<Mutex<()>>,
    sa: Arc<Semaphore>,
    sb: Arc<Semaphore>,
) {
    drop(a.lock().unwrap());

    sa.acquire().release();

    drop(sb.acquire());

    let mut flag = a.lock().unwrap();
    let _b_guard = b.lock().unwrap();
    *flag = true;
    drop(_b_guard);
    drop(flag);
}
