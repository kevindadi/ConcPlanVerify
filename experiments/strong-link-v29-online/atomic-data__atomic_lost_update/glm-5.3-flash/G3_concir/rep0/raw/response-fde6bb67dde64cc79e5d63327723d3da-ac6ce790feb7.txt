use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: &AtomicI32) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let next = old + 1;
        let observed = c.compare_exchange(old, next, Ordering::SeqCst, Ordering::SeqCst);
        match observed {
            Ok(_) => break,
            Err(_) => continue,
        }
    }
}

fn w2(c: &AtomicI32) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let next = old + 1;
        let observed = c.compare_exchange(old, next, Ordering::SeqCst, Ordering::SeqCst);
        match observed {
            Ok(_) => break,
            Err(_) => continue,
        }
    }
}

fn main() {
    let c = Arc::new(AtomicI32::new(0));

    let c1 = Arc::clone(&c);
    let h1 = thread::spawn(move || w1(&c1));

    let c2 = Arc::clone(&c);
    let h2 = thread::spawn(move || w2(&c2));

    h1.join().expect("w1 panicked");
    h2.join().expect("w2 panicked");

    let done = c.load(Ordering::SeqCst);
    println!("DONE done={}", done / 2);
}
