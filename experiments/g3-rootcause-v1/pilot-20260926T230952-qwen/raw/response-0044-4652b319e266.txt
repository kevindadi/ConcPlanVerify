use std::sync::{Arc, Mutex, Condvar};
use std::thread;

fn main() {
    let m = Arc::new(Mutex::new(false));
    let cv = Arc::new(Condvar::new());

    let m_waiter = Arc::clone(&m);
    let cv_waiter = Arc::clone(&cv);
    let waiter_handle = thread::spawn(move || {
        // Function: main::waiter
        // s1: mutex_lock main::m
        let mut guard = m_waiter.lock().unwrap();
        
        // s2: branch ready == true (loop condition)
        while !*guard {
            // s3: condvar_wait main::cv on main::m
            guard = cv_waiter.wait(guard).unwrap();
            // s4: goto s2 (handled by loop)
        }
        
        // s5: mutex_unlock main::m
        drop(guard);
        // s6: return
    });

    let m_notifier = Arc::clone(&m);
    let cv_notifier = Arc::clone(&cv);
    let notifier_handle = thread::spawn(move || {
        // Function: main::notifier
        // s1: mutex_lock main::m
        let mut guard = m_notifier.lock().unwrap();
        
        // s2: write_shared main::ready = true
        *guard = true;
        
        // s3: condvar_notify main::cv
        cv_notifier.notify_one();
        
        // s4: mutex_unlock main::m
        drop(guard);
        // s5: return
    });

    waiter_handle.join().unwrap();
    notifier_handle.join().unwrap();

    let ready = *m.lock().unwrap();
    println!("DONE ready={}", ready);
}
