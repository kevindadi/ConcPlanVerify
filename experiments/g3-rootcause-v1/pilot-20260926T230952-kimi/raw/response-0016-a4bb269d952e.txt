use std::sync::{Arc, Mutex};
use std::sync::mpsc::{sync_channel, Receiver, SyncSender};
use std::thread;

fn sender(m: Arc<Mutex<()>>, ch: SyncSender<i32>) {
    {
        let _guard = m.lock().unwrap();
    }
    ch.send(1).unwrap();
    ch.send(2).unwrap();
}

fn receiver(m: Arc<Mutex<()>>, ch: Receiver<i32>) {
    {
        let _guard = m.lock().unwrap();
    }
    let _ = ch.recv().unwrap();
    let _ = ch.recv().unwrap();
}

fn main() {
    let m = Arc::new(Mutex::new(()));
    let (tx, rx) = sync_channel::<i32>(1);

    let m_sender = Arc::clone(&m);
    let m_receiver = Arc::clone(&m);

    let sender_handle = thread::spawn(move || sender(m_sender, tx));
    let receiver_handle = thread::spawn(move || receiver(m_receiver, rx));

    sender_handle.join().unwrap();
    receiver_handle.join().unwrap();

    println!("DONE done=1");
}
