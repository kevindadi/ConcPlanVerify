use concir_sync::Semaphore;

fn w1(s: &Semaphore) {
    let mut i = 0;
    while i < 2 {
        let permit = s.acquire();
        permit.release();
        i += 1;
    }
}

fn w2(s: &Semaphore) {
    let mut i = 0;
    while i < 2 {
        let permit = s.acquire();
        permit.release();
        i += 1;
    }
}

fn main() {
    let s = Semaphore::new(1);
    let s_w1 = s.clone();
    let s_w2 = s.clone();

    let h1 = std::thread::spawn(move || w1(&s_w1));
    let h2 = std::thread::spawn(move || w2(&s_w2));

    h1.join().unwrap();
    h2.join().unwrap();

    println!("DONE done=1");
}
