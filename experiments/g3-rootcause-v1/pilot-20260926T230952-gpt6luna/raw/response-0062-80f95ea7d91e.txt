use concir_sync::Semaphore;
use std::hint::black_box;
use std::sync::{Arc, Mutex};
use std::thread;

fn a(
    a_lock: Arc<Mutex<i32>>,
    b_lock: Arc<Mutex<()>>,
    sa: Arc<Semaphore>,
    sb: Arc<Semaphore>,
) {
    let sa_permit = sa.acquire();

    let _guard = a_lock.lock().unwrap();
    sa_permit.release();
    drop(_guard);

    let _sb_permit = sb.acquire();

    let mut flag = a_lock.lock().unwrap();
    let _b_guard = b_lock.lock().unwrap();
    *flag = 1;
}

fn b(
    a_lock: Arc<Mutex<i32>>,
    b_lock: Arc<Mutex<()>>,
    sa: Arc<Semaphore>,
    sb: Arc<Semaphore>,
) {
    let sb_permit = sb.acquire();

    let _sa_permit = sa.acquire();
    let _guard = a_lock.lock().unwrap();
    sb_permit.release();
    drop(_guard);

    let mut flag = a_lock.lock().unwrap();
    let _b_guard = b_lock.lock().unwrap();
    *flag = 1;
}

fn bystander() {
    let mut i = 0;
    loop {
        i = if i == 0 { 1 } else { 0 };
        black_box(i);
    }
}

fn main() {
    let a_lock = Arc::new(Mutex::new(0));
    let b_lock = Arc::new(Mutex::new(()));
    let sa = Semaphore::new(1);
    let sb = Semaphore::new(1);

    let a_worker = {
        let a_lock = Arc::clone(&a_lock);
        let b_lock = Arc::clone(&b_lock);
        let sa = Arc::clone(&sa);
        let sb = Arc::clone(&sb);
        thread::spawn(move || a(a_lock, b_lock, sa, sb))
    };

        let a_lock = Arc::clone(&a_lock);
        let b_lock = Arc::clone(&b_lock);
        let sa = Arc::clone(&sa);
        let sb = Arc::clone(&sb);
        thread::spawn(move || b(a_lock, b_lock, sa, sb))
    };

    let _bystander_task = thread::spawn(bystander);

    a_worker.join().unwrap();
    b_worker.join().unwrap();

    let flag = *a_lock.lock().unwrap();
    println!("DONE a={} b={}", flag, flag);
}
