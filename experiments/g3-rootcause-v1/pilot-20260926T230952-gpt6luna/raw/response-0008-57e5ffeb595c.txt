use std::sync::{Arc, Mutex};
use std::thread;

fn w1(m: Arc<Mutex<i32>>) {
    let mut c = m.lock().unwrap();
    *c += 1;
}

fn w2(m: Arc<Mutex<i32>>) {
    let mut c = m.lock().unwrap();
    *c += 1;
}

fn main() {
    let m = Arc::new(Mutex::new(0));

    let worker1 = {
        let m = Arc::clone(&m);
        thread::spawn(move || w1(m))
    };
    let worker2 = {
        let m = Arc::clone(&m);
        thread::spawn(move || w2(m))
    };

    worker1.join().unwrap();
    worker2.join().unwrap();

    println!("DONE done=1");
}
