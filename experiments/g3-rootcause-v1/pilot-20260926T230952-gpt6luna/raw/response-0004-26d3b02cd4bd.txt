use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicI32>) {
    let mut old = 0;
    let mut seen;

    loop {
        old = c.load(Ordering::SeqCst);
        seen = match c.compare_exchange(old, old + 1, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(previous) | Err(previous) => previous,
        };
        if seen == old {
            return;
        }
    }
}

fn w2(c: Arc<AtomicI32>) {

    }
}

fn main() {
