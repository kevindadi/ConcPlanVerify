use std::sync::atomic::{AtomicI64, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicI64>) {
    let mut old: i64;
    let mut new: i64;
    loop {
        old = c.load(Ordering::SeqCst);
        new = old + 1;
        old = c.compare_exchange(old, new, Ordering::SeqCst, Ordering::SeqCst)
            .unwrap_or_else(|v| v);
        if old == new - 1 {
            return;
        }
    }
}

fn w2(c: Arc<AtomicI64>) {
    let mut old: i64;
    let mut new: i64;
    loop {
        old = c.load(Ordering::SeqCst);
        new = old + 1;
        old = c.compare_exchange(old, new, Ordering::SeqCst, Ordering::SeqCst)
            .unwrap_or_else(|v| v);
        if old == new - 1 {
            return;
        }
    }
}

fn main() {
    let c = Arc::new(AtomicI64::new(0));

    let c1 = Arc::clone(&c);
    let c2 = Arc::clone(&c);

    let h1 = thread::spawn(move || w1(c1));
    let h2 = thread::spawn(move || w2(c2));

    h1.join().unwrap();
    h2.join().unwrap();

    println!("DONE done=1");
}
