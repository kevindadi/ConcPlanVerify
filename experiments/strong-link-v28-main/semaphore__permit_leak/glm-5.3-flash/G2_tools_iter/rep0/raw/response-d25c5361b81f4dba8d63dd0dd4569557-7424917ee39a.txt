```rust
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

// Shared resource: s (semaphore) — a simple counting semaphore
// built from a Mutex and a Condvar so that a waiting worker
// releases the lock while blocked (R5) and can be woken by the
// releasing worker.
struct Semaphore {
    permits: Mutex<usize>,
    available: Condvar,
}

impl Semaphore {
    fn new(initial: usize) -> Self {
        Semaphore {
            permits: Mutex::new(initial),
            available: Condvar::new(),
        }
    }

    // Acquire one permit; block (without holding the lock) until one
    // is available.
    fn acquire(&self) {
        let mut count = self.permits.lock().unwrap();
        while *count == 0 {
            count = self.available.wait(count).unwrap();
        }
        *count -= 1;
    }

    // Release one permit and wake a waiting worker.
    fn release(&self) {
        let mut count = self.permits.lock().unwrap();
        *count += 1;
        self.available.notify_one();
    }
}

// Worker role: w1 / w2
// Each worker acquires one permit, performs its work while holding
// it, and releases the permit before finishing (R3, R4).
fn worker(name: &'static str, s: Arc<Semaphore>) {
    s.acquire();
    println!("{}: working with permit", name);
    // Simulated work — the permit is held for the duration.
    s.release();
    println!("{}: released permit", name);
}

fn main() {
    // Supervising task: launches w1 and w2 and waits for both (R1).
    // The shared semaphore s starts with exactly one permit (R2).
    let s = Arc::new(Semaphore::new(1));

    let s1 = Arc::clone(&s);
    let h1 = thread::spawn(move || worker("w1", s1));

    let s2 = Arc::clone(&s);
    let h2 = thread::spawn(move || worker("w2", s2));

    h1.join().unwrap();
    h2.join().unwrap();

    // Both workers have finished; the permit pool is back to one.
    let final_permits = *s.permits.lock().unwrap();
    println!("DONE permits={}", final_permits);
}
```

Notes on how the requirements are met:

- **R1**: `main` spawns the two workers (`w1`, `w2`) and joins both handles before proceeding.
- **R2**: `Semaphore::new(1)` initializes the shared pool with exactly one permit; it is shared via `Arc`.
- **R3/R4**: Each worker calls `acquire()` before its work and `release()` after, so the critical sections never overlap and every acquired permit is returned.
- **R5**: The `Condvar::wait` call atomically unlocks the mutex while blocking, so the worker holding the permit is never prevented from releasing it; `release()` calls `notify_one()` to wake the waiter.
- **R6**: The `while *count == 0` loop guards against spurious wakeups, and the mutex/condvar protocol guarantees no deadlock or lost wakeup for any interleaving.
- **R7**: After both joins, the program prints exactly `DONE permits=1` and exits.
