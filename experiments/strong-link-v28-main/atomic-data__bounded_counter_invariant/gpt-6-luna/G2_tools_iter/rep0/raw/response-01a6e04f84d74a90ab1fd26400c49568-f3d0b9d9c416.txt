use std::sync::{Arc, Mutex};
use std::thread;

fn w1(m: Arc<Mutex<u8>>) {
    let mut c = m.lock().unwrap();
    assert!(*c < 2);
    *c += 1;
}

fn w2(m: Arc<Mutex<u8>>) {
    let mut c = m.lock().unwrap();
    assert!(*c < 2);
    *c += 1;
}

fn main() {
    let m = Arc::new(Mutex::new(0u8));

    let t1 = {
        let m = Arc::clone(&m);
        thread::spawn(move || w1(m))
    };
    let t2 = {
        let m = Arc::clone(&m);
        thread::spawn(move || w2(m))
    };

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
