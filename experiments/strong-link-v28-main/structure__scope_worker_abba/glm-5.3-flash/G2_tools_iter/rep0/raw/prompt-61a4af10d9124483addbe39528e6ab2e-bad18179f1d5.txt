//! Two workers (w1, w2) contend for two shared mutexes (a, b).
//!
//! R1: main spawns two workers that contend for mutexes a and b.
//! R2: each worker takes a then b, so it holds both simultaneously.
//! R3: each worker releases both mutexes (guards drop) when its work is done.
//! R4: std::sync::Mutex guarantees at most one holder per mutex.
//! R5: both workers lock in the same order (a, then b) -> no wait cycle.
//! R6: lock() blocks until the mutex becomes free.
//! R7: consistent lock ordering means every interleaving terminates.
//! R8: main joins both workers; the group finishes only after both complete.
//! R9: the program prints exactly `DONE done=1` and exits.

use std::sync::{Arc, Mutex};
use std::thread;

/// Body executed by each worker. `name` is only for documentation purposes;
/// workers produce no output so the program's sole printed line is `DONE done=1`.
fn worker(name: &'static str, a: Arc<Mutex<()>>, b: Arc<Mutex<()>>) {
    // R5/R6: take a first, then b — same order as the other worker, blocking
    // until each mutex is free. No cycle can form, so this always terminates.
    let _guard_a: MutexGuard = a.lock().unwrap();

    // R2: after this line the worker holds both a and b at the same time.
    let _guard_b: MutexGuard = b.lock().unwrap();

    // Simulate work while holding both mutexes. (No output: R9 requires the
    // program to print exactly one line.)
    let _ = name;
    std::hint::black_box(());

    // R3: work is finished; both guards drop here, releasing b then a.
}

type MutexGuard = std::sync::MutexGuard<'static, ()>;

fn main() {
    // Shared resources a and b.
    let a: Arc<Mutex<()>> = Arc::new(Mutex::new(()));
    let b: Arc<Mutex<()>> = Arc::new(Mutex::new(()));

    // R1: start the group of two workers, w1 and w2.
    let mut handles = Vec::new();
    for name in ["w1", "w2"] {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        handles.push(thread::spawn(move || worker(name, a, b)));
    }

    // R8: the group finishes only after both workers have completed.
    for handle in handles {
        handle.join().expect("worker panicked");
    }

    // R9: print exactly this line, then exit.
    println!("DONE done=1");
}
