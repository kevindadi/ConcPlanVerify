```rust
use std::sync::mpsc::sync_channel;
use std::sync::{Arc, Mutex};
use std::thread;

// State protected by the shared lock m.
#[derive(Default)]
struct State {
    touches: u32, // evidence both roles use the lock
    done: bool,   // set by receiver after taking both values
}

fn main() {
    // ch: a channel that can hold at most one value.
    let (tx, rx) = sync_channel::<i32>(1);
    // m: the one shared lock.
    let m = Arc::new(Mutex::new(State::default()));

    // ---- sender role ----
    let m_sender = Arc::clone(&m);
    let sender = thread::spawn(move || {
        for v in 1..=2 {
            // Take the lock only briefly, and ALWAYS release it before
            // waiting on the channel (R5: no hold-and-wait on m + ch).
            {
                let mut s = m_sender.lock().unwrap();
                s.touches += 1;
            } // lock released here
            // Blocks when the channel is full (R4). Since capacity is 1,
            // the second send cannot proceed until the receiver has taken
            // the first value (R7).
            tx.send(v).unwrap();
        }
    });

    // ---- receiver role ----
    let m_receiver = Arc::clone(&m);
    let receiver = thread::spawn(move || {
        let mut got = 0;
        while got < 2 {
            // Same discipline: lock is released before any channel wait.
            {
                let mut s = m_receiver.lock().unwrap();
                s.touches += 1;
            } // lock released here
            // Blocks when the channel is empty (R4).
            if rx.recv().is_ok() {
                got += 1;
            }
        }
        let mut s = m_receiver.lock().unwrap();
        s.done = true;
    });

    sender.join().unwrap();
    receiver.join().unwrap();

    let done = {
        let s = m.lock().unwrap();
        s.done
    };

    // R8: print exactly this line, then exit.
    println!("DONE done={}", if done { 1 } else { 0 });
}
```

Why this satisfies the requirements:

- **R1**: `main` spawns the `sender` and `receiver` threads and both run concurrently.
- **R2**: `ch` is a `sync_channel(1)` (capacity one) and `m` is a single `Mutex` that both roles lock.
- **R3**: sender passes `1` then `2`; receiver takes two values; both threads then finish and are joined.
- **R4**: `sync_channel`'s `send` blocks while the channel is full and `recv` blocks while it is empty.
- **R5**: Both roles acquire `m` only inside short scoped blocks and drop it *before* calling `send`/`recv`, so neither role ever waits on the channel while holding the lock — the hold-and-wait condition needed for deadlock is impossible.
- **R6**: Since no thread ever holds `m` while blocked on `ch`, every interleaving terminates; both threads complete and `main` exits.
- **R7**: With capacity 1, the sender's second `send` cannot complete until the receiver's first `recv` has drained the channel.
- **R8**: The receiver sets `done` under `m` after taking both values; `main` reads it and prints exactly `DONE done=1`.
