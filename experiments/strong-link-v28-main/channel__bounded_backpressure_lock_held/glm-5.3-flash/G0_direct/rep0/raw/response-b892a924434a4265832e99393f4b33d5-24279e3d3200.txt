```rust
use std::sync::mpsc::sync_channel;
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    // Shared lock m, used occasionally by both roles.
    let m = Arc::new(Mutex::new(()));

    // Channel ch with capacity 1: it can hold at most one value.
    let (tx, rx) = sync_channel::<i32>(1);

    // Sender role: passes two values in order (1, then 2).
    let m_sender = Arc::clone(&m);
    let sender = thread::spawn(move || {
        for v in 1..=2 {
            // Take the shared lock only briefly, and release it
            // BEFORE waiting on the channel (R5).
            {
                let _guard = m_sender.lock().unwrap();
                // brief critical section; no channel wait while holding m
            }
            // If the channel is full, this waits until the receiver
            // has taken the previous value (R4, R7).
            tx.send(v).unwrap();
        }
        // Sender finishes after passing both values (R3).
    });

    // Receiver role: takes two values.
    let m_receiver = Arc::clone(&m);
    let receiver = thread::spawn(move || {
        for _ in 0..2 {
            // Take the shared lock only briefly, and release it
            // BEFORE waiting on the channel (R5).
            {
                let _guard = m_receiver.lock().unwrap();
                // brief critical section; no channel wait while holding m
            }
            // If the channel is empty, this waits until the sender
            // has put a value (R4).
            let _v = rx.recv().unwrap();
        }
        // Receiver finishes after taking both values (R3).
    });

    // Both roles run at the same time and both must finish (R1, R6).
    sender.join().unwrap();
    receiver.join().unwrap();

    // Exactly one line of output, then exit (R8).
    println!("DONE done=1");
}
```

Notes on how the requirements are met:

- **R1**: `main` spawns `sender` and `receiver` threads that run concurrently.
- **R2**: `ch` is a `sync_channel` with capacity 1 (holds at most one value); `m` is a shared `Mutex` both roles occasionally lock.
- **R3**: The sender sends values `1` and `2` in order; the receiver receives twice; both threads then end.
- **R4**: `sync_channel(1)` makes `send` block while the channel is full and `recv` block while it is empty.
- **R5**: Each role acquires `m` in a short scoped block and drops the guard before calling `send`/`recv`, so no role ever waits on the channel while holding `m`.
- **R6**: The channel capacity of 1 plus the send/recv pairing guarantees every interleaving terminates; both `join` calls complete.
- **R7**: With capacity 1, the second `send` cannot proceed until the receiver has taken the first value.
- **R8**: The only output is the single line `DONE done=1`, printed after both threads finish.
