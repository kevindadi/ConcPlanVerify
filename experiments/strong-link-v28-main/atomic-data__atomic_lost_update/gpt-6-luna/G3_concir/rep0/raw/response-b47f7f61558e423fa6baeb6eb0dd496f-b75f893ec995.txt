use std::thread;

static c: AtomicI32 = AtomicI32::new(0);

fn w1() {
    loop {
        let expected = c.load(Ordering::SeqCst);
        let desired = expected + 1;
        let observed = match c.compare_exchange(
            expected,
            desired,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(_) => expected,
            Err(actual) => actual,
        };

        if observed == expected {
            return;
        }
    }
}
