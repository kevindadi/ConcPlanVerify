use std::sync::{
    atomic::{AtomicUsize, Ordering},
    Arc, Mutex,
};
use std::thread;

fn t1(a: Arc<Mutex<()>>, b: Arc<Mutex<()>>, work: Arc<AtomicUsize>) {
    let first = a.lock().unwrap();
    let second = b.lock().unwrap();

    work.fetch_add(1, Ordering::Relaxed);

    drop(second);
    drop(first);
}

fn t2(b: Arc<Mutex<()>>, c: Arc<Mutex<()>>, work: Arc<AtomicUsize>) {
    let first = b.lock().unwrap();
    let second = c.lock().unwrap();

    work.fetch_add(1, Ordering::Relaxed);

    drop(second);
    drop(first);
}

fn t3(a: Arc<Mutex<()>>, c: Arc<Mutex<()>>, work: Arc<AtomicUsize>) {
    let first = a.lock().unwrap();
    let second = c.lock().unwrap();

    work.fetch_add(1, Ordering::Relaxed);

    drop(second);
    drop(first);
}

fn main() {
    let a = Arc::new(Mutex::new(()));
    let b = Arc::new(Mutex::new(()));
    let c = Arc::new(Mutex::new(()));
    let work = Arc::new(AtomicUsize::new(0));

    let worker1 = {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        let work = Arc::clone(&work);
        thread::spawn(move || t1(a, b, work))
    };

    let worker2 = {
        let b = Arc::clone(&b);
        let c = Arc::clone(&c);
        let work = Arc::clone(&work);
        thread::spawn(move || t2(b, c, work))
    };

    let worker3 = {
        let a = Arc::clone(&a);
        let c = Arc::clone(&c);
        let work = Arc::clone(&work);
        thread::spawn(move || t3(a, c, work))
    };

    worker1.join().unwrap();
    worker2.join().unwrap();
    worker3.join().unwrap();

    println!("DONE done=1");
}
