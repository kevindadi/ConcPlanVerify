use std::thread;

fn s1(ch: SyncSender<()>) {

fn r(ch: Receiver<()>) {
    ch.recv().unwrap();

    let (sender, receiver) = ch;

    let s1_task = thread::spawn(move || s1(sender));
    let r_task = thread::spawn(move || r(receiver));

}
