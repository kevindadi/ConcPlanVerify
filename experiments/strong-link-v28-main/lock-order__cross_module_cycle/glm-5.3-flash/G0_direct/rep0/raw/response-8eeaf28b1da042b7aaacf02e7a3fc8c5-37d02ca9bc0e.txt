```rust
// Design overview:
// - module1 owns shared resource `a` (a Mutex).
// - module2 owns shared resource `b` (a Mutex).
// - Task t1 runs in module1; task t2 runs in module2.
// - Each task declares its dependency on the resource owned by the other
//   module by receiving that module's lock as an explicit parameter.
// - Both tasks acquire the locks in one fixed global order (a, then b),
//   so no interleaving can produce circular waiting: deadlock is impossible
//   and every schedule terminates.
// - Each task explicitly releases both locks before it finishes.
// - The starting thread (main) launches both tasks and finishes only after
//   both have been joined.

mod module1 {
    use std::sync::Mutex;

    // Resource `a` is owned by module1.
    pub static A: Mutex<()> = Mutex::new(());

    // Task t1 runs in module1. It depends on resource `b`, owned by module2,
    // which it declares by taking `b` as a parameter.
    pub fn t1(b: &'static Mutex<()>) {
        // Fixed global order: a, then b.
        let ga = A.lock().unwrap();
        let gb = b.lock().unwrap();

        // Work while holding both resources at the same time.
        let _work = ga;
        let _work = gb;

        // Release each resource before finishing.
        drop(_work);
        drop(_work);
    }
}

mod module2 {
    use std::sync::Mutex;

    // Resource `b` is owned by module2.
    pub static B: Mutex<()> = Mutex::new(());

    // Task t2 runs in module2. It depends on resource `a`, owned by module1,
    // which it declares by taking `a` as a parameter.
    pub fn t2(a: &'static Mutex<()>) {
        // Same fixed global order: a, then b.
        let ga = a.lock().unwrap();
        let gb = B.lock().unwrap();

        // Work while holding both resources at the same time.
        let _work = ga;
        let _work = gb;

        // Release each resource before finishing.
        drop(_work);
        drop(_work);
    }
}

fn main() {
    // Starting thread launches both tasks.
    let h1 = std::thread::spawn(|| module1::t1(&module2::B));
    let h2 = std::thread::spawn(|| module2::t2(&module1::A));

    // The starting thread finishes only after both tasks have finished.
    h1.join().unwrap();
    h2.join().unwrap();

    println!("DONE done=1");
}
```
