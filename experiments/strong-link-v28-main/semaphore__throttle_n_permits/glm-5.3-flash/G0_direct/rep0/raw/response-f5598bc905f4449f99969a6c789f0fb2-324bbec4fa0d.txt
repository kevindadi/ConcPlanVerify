```rust
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

// Shared resource: s (counting permit pool)
struct Semaphore {
    permits: Mutex<usize>,
    cv: Condvar,
}

impl Semaphore {
    fn new(initial: usize) -> Self {
        Semaphore {
            permits: Mutex::new(initial),
            cv: Condvar::new(),
        }
    }

    fn acquire(&self) {
        let mut count = self.permits.lock().unwrap();
        while *count == 0 {
            // Wait while a holder remains able to release (R5)
            count = self.cv.wait(count).unwrap();
        }
        *count -= 1;
    }

    fn release(&self) {
        let mut count = self.permits.lock().unwrap();
        *count += 1;
        self.cv.notify_one();
    }
}

// Worker role: w1
fn w1(s: &Arc<Semaphore>) {
    s.acquire();
    // perform work
    s.release();
}

// Worker role: w2
fn w2(s: &Arc<Semaphore>) {
    s.acquire();
    // perform work
    s.release();
}

// Worker role: w3
fn w3(s: &Arc<Semaphore>) {
    s.acquire();
    // perform work
    s.release();
}

fn main() {
    // Supervising task: launches three workers and waits for all of them (R1)
    // Pool begins with exactly two permits (R2)
    let s = Arc::new(Semaphore::new(2));

    let s1 = Arc::clone(&s);
    let t1 = thread::spawn(move || w1(&s1));
    let s2 = Arc::clone(&s);
    let t2 = thread::spawn(move || w2(&s2));
    let s3 = Arc::clone(&s);
    let t3 = thread::spawn(move || w3(&s3));

    t1.join().unwrap();
    t2.join().unwrap();
    t3.join().unwrap();

    // R7: print exactly this line and exit
    println!("DONE done=1");
}
```

Notes on how the requirements are met:

- **R1**: `main` acts as the supervising task, spawning `w1`, `w2`, `w3` and joining all three threads.
- **R2**: `s` is a single `Arc<Semaphore>` shared by all workers, initialized with exactly `2` permits.
- **R3**: Each worker calls `acquire()`, does its work, then calls `release()` before finishing.
- **R4**: The counter never goes below zero, so at most two workers hold permits concurrently; the third blocks in `acquire()` until a release occurs.
- **R5**: A waiting worker blocks on the `Condvar` while holding no lock, so any permit holder can still lock the mutex and call `release()`, which notifies the waiter.
- **R6**: Every blocking wait is paired with a guaranteed release from a holder, so all schedules and interleavings terminate.
- **R7**: After all workers finish, the program prints exactly `DONE done=1` and exits.
