// Roles: w1, w2 (workers), compute (sequential helper).
// Shared resources: m (the shared mutex), acc (the shared counter it guards).

use std::sync::{Arc, Mutex};
use std::thread;

// Sequential helper routine: performs only local computation,
// touches no shared state.
fn compute(seed: u64) -> u64 {
    let mut x = seed;
    for i in 0..64u64 {
        x = x.wrapping_mul(6364136223846793005).wrapping_add(i + 1);
    }
    x
}

// Worker body (used by both w1 and w2).
// R2: take the shared mutex m, call compute (local only), then update
//     the shared counter acc while still holding m.
// R3: release m before the worker finishes.
// R4: guaranteed by the mutex — at most one holder at any moment.
// R5: lock() blocks until the mutex becomes free.
// R6: no nested locks, no recursion, no unbounded loops => every
//     interleaving terminates.
fn worker(m: &Mutex<u64>) {
    // Take m; if another worker holds it, wait until it is free (R5).
    let mut acc = m.lock().unwrap();

    // Sequential, purely local computation while holding m (R2).
    let _local = compute(7);

    // Update the shared counter acc while still holding m (R2).
    *acc = acc.wrapping_add(1);

    // Release m before finishing (R3).
    drop(acc);
}

fn main() {
    // R1: main starts a group of two workers sharing one mutex m,
    //     which guards the shared counter acc (initialized to 0).
    let m = Arc::new(Mutex::new(0u64));

    let m1 = Arc::clone(&m);
    let m2 = Arc::clone(&m);

    let w1 = thread::spawn(move || worker(&m1));
    let w2 = thread::spawn(move || worker(&m2));

    // R7: wait for both workers to complete.
    w1.join().expect("w1 panicked");
    w2.join().expect("w2 panicked");

    // R8: print exactly the required line, then exit.
    let done = 1;
    println!("DONE done={}", done);
}
