```rust
use std::sync::{Arc, Mutex};
use std::thread;

struct Acc {
    acc: i32,
}

fn compute() -> i32 {
    let mut tmp = 0;
    tmp += 1;
    tmp
}

fn w1(m: Arc<Mutex<Acc>>) {
    let mut guard = m.lock().unwrap();
    compute();
    guard.acc += 1;
    drop(guard);
}

fn w2(m: Arc<Mutex<Acc>>) {
    let mut guard = m.lock().unwrap();
    compute();
    guard.acc += 1;
    drop(guard);
}

fn main() {
    let m = Arc::new(Mutex::new(Acc { acc: 0 }));

    let m1 = Arc::clone(&m);
    let t1 = thread::spawn(move || w1(m1));

    let m2 = Arc::clone(&m);
    let t2 = thread::spawn(move || w2(m2));

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
```
