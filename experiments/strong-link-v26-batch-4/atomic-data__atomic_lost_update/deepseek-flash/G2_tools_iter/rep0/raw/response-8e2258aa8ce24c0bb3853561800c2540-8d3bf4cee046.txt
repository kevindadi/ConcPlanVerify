use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicUsize>) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let new = old + 1;
        match c.compare_exchange(old, new, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(_) => break,
            Err(_) => continue,
        }
    }
}

fn w2(c: Arc<AtomicUsize>) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let new = old + 1;
        match c.compare_exchange(old, new, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(_) => break,
            Err(_) => continue,
        }
    }
}

fn main() {
    let c = Arc::new(AtomicUsize::new(0));
    let c1 = Arc::clone(&c);
    let c2 = Arc::clone(&c);

    let t1 = thread::spawn(move || w1(c1));
    let t2 = thread::spawn(move || w2(c2));

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
