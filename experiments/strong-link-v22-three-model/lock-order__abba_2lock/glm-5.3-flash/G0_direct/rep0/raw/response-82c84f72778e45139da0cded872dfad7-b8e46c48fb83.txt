```rust
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let a = Arc::new(Mutex::new(()));
    let b = Arc::new(Mutex::new(()));

    let t1 = {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        thread::spawn(move || {
            let _a = a.lock().unwrap();
            let _b = b.lock().unwrap();
            // critical work holding both locks
        })
    };

    let t2 = {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        thread::spawn(move || {
            let _a = a.lock().unwrap();
            let _b = b.lock().unwrap();
            // critical work holding both locks
        })
    };

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE t1=1 t2=1");
}
```

Both workers acquire locks `a` and `b` in the same order (a then b), so no circular wait is possible (R5/R8). Locks are acquired with blocking `lock()` calls (R4), held simultaneously during the critical work (R3), and released via RAII guard drops before the worker finishes (R7). The main thread joins both workers before printing and exiting (R6, R9).
