use std::sync::{Arc, Mutex};

fn t1(a: &Mutex<()>, b: &Mutex<()>) {
    let _ga = a.lock().unwrap();
    let _gb = b.lock().unwrap();
    drop(_gb);
    drop(_ga);
}

fn t2(a: &Mutex<()>, b: &Mutex<()>) {
    let _ga = a.lock().unwrap();
    let _gb = b.lock().unwrap();
    drop(_gb);
    drop(_ga);
}

fn main() {
    let a = Arc::new(Mutex::new(()));
    let b = Arc::new(Mutex::new(()));

    let h1 = {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        std::thread::spawn(move || t1(&a, &b))
    };
    let h2 = {
        let a = Arc::clone(&a);
        let b = Arc::clone(&b);
        std::thread::spawn(move || t2(&a, &b))
    };

    h1.join().unwrap();
    h2.join().unwrap();

    println!("DONE t1=1 t2=1");
}
