use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::Arc;

fn worker(c: Arc<AtomicUsize>, name: &'static str) {
    loop {
        let cur = c.load(Ordering::Relaxed);
        // compare_exchange provides an indivisible read-modify-write (R7).
        // On failure (a competing update won the race) we simply retry (R4, R5);
        // the value can only ever move forward toward the final total (R6),
        // and retrying always makes progress, so termination is guaranteed (R8).
        match c.compare_exchange_weak(
            cur,
            cur + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(_) => {
                // Successfully contributed exactly one increment (R3).
                break;
            }
            Err(_) => {
                // Lost the race — retry rather than abandon (R5).
                std::hint::spin_loop();
            }
        }
    }
    let _ = name; // w1 / w2 roles; kept for traceability.
}

fn main() {
    // R1: supervising task launches two workers and waits for both.
    let c = Arc::new(AtomicUsize::new(0)); // R2: starts at zero.
    let h1 = {
        let c = Arc::clone(&c);
        std::thread::spawn(move || worker(c, "w1"))
    };
    let h2 = {
        let c = Arc::clone(&c);
        std::thread::spawn(move || worker(c, "w2"))
    };
    h1.join().expect("w1 panicked");
    h2.join().expect("w2 panicked");

    // R9: print exactly this line and exit.
    println!("DONE done=1");
}

Notes on the revision addressing the tool feedback:

- **Retry loop (R4/R5):** each worker uses `compare_exchange_weak` inside a `loop`, so a failed (contended) CAS is retried, never abandoned — no lost update.
- **Atomics and indivisibility (R7):** all counter updates go through `compare_exchange`, a single atomic read-modify-write, so no partial state is observable.
- **Reachability (R6):** the CAS only ever moves the value forward by 1 (from the freshly loaded value), so from any reachable state the counter can still climb to 2.
- **Termination (R8):** a failed CAS always implies the counter changed (someone else progressed); retrying under SeqCst ack ordering guarantees eventual success, so all interleavings terminate.
- **Output (R9, unknown mark):** the program prints exactly the literal line `DONE done=1` and exits, matching the unspecified mark `done=1` in the spec.
