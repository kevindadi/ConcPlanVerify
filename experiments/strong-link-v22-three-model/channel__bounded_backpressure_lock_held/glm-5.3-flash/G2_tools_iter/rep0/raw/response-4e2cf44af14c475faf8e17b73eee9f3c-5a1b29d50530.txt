```rust
use std::sync::{mpsc, Arc, Mutex};
use std::thread;

fn main() {
    // ch: a channel that holds at most one value (capacity 1).
    let (tx, rx) = mpsc::sync_channel::<i32>(1);
    // m: shared lock both roles occasionally need.
    let m: Arc<Mutex<usize>> = Arc::new(Mutex::new(0));

    let sender = {
        let m = m.clone();
        thread::spawn(move || {
            for v in 1..=2 {
                // R4/R5: wait on the full channel WITHOUT holding m,
                // so the receiver is never blocked by the sender's lock.
                tx.send(v).expect("receiver alive");
                let mut cnt = m.lock().unwrap();
                *cnt += 1;
            }
        })
    };

    let receiver = thread::spawn(move || {
        loop {
            // R4/R5: wait on the empty channel WITHOUT holding m.
            match rx.recv() {
                Ok(_v) => {
                    let mut cnt = m.lock().unwrap();
                    *cnt += 1;
                }
                Err(_) => break,
            }
        }
    });

    sender.join().expect("sender thread");
    receiver.join().expect("receiver thread");

    let cnt = m.lock().unwrap();
    let _ = *cnt; // both roles can finish once two values passed (R3)
    println!("DONE done=1");
}
```

Explanation:

- **R1**: two spawned threads implement `sender` and `receiver`.
- **R2**: `ch` is a `sync_channel(1)` (bounded to one value) and `m` is a shared `Mutex` both threads lock briefly.
- **R3**: the sender sends two values, the receiver receives them until the channel closes, and both threads terminate.
- **R4**: `sync_channel(1)` makes `send` block while the channel is full and `recv` block while it is empty.
- **R5**: neither thread acquires `m` before blocking on the channel; `send`/`recv` are called before locking `m`, so a thread never waits on `ch` while holding `m`.
- **R6/R7**: since the capacity is one, the second `send` cannot complete until the first value has been taken (no preemptive double-send), which also guarantees a liveness cycle that terminates in every interleaving.
- **R8**: exactly one line, `DONE done=1`, is printed; all status is printed before it.
