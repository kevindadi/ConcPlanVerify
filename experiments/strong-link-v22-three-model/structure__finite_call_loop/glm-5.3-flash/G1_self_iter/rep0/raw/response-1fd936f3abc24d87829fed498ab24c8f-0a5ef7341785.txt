// helper: a pure auxiliary routine with no shared state (R2).
// Each call runs to completion before the next call begins (R3):
// calls are sequential in one task, so no interleaving is possible (R4).

fn helper(x: u64) -> u64 {
    x + 1
}

fn main() {
