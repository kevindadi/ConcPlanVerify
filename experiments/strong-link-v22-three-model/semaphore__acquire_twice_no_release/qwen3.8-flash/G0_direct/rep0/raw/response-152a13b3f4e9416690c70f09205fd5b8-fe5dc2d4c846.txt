```rust
use std::sync::{Arc, Barrier};
use std::thread;

// We need a semaphore with exactly 1 permit.
// Rust doesn't have a built-in counting semaphore in std, but we can simulate it
// using a Mutex and Condvar, or use a simpler approach with an atomic counter and spin-wait?
// Actually, the requirement says "counting permit pool that begins with exactly one permit".
// This is essentially a binary semaphore / mutex-like behavior where only one worker can hold the permit at a time.
// 
// Let's implement a simple semaphore using Mutex<isize> and Condvar.
// Or even simpler: since there's exactly 1 permit, this behaves like a lock, but workers may acquire/release multiple times.
// The key constraints:
// R3: Workers never work at the same time (mutual exclusion via the permit).
// R4: Each worker acquires/releases equally many times.
// R5: While waiting, the holder must be able to release.
// R6: All schedules terminate.
// R7: Print "DONE done=1" and exit.

// Let's create a Semaphore struct.

struct Semaphore {
    permits: std::sync::Mutex<usize>,
    condvar: std::sync::Condvar,
}

impl Semaphore {
    fn new(permits: usize) -> Self {
        Semaphore {
            permits: std::sync::Mutex::new(permits),
            condvar: std::sync::Condvar::new(),
        }
    }

    fn acquire(&self) {
        let mut guard = self.permits.lock().unwrap();
        while *guard == 0 {
            guard = self.condvar.wait(guard).unwrap();
        }
        *guard -= 1;
    }

    fn release(&self) {
        let mut guard = self.permits.lock().unwrap();
        *guard += 1;
        self.condvar.notify_one();
    }
}

fn main() {
    // Shared semaphore with 1 permit
    let s = Arc::new(Semaphore::new(1));

    // We need two worker threads w1 and w2
    // Each worker must acquire the permit, do some work, then release it.
    // They may acquire more than once, but must release as many times as acquired.
    
    // For simplicity and to ensure termination and correct output, let each worker:
    // - Acquire the permit once
    // - Do some trivial work
    // - Release the permit once
    
    // But wait, if both just acquire-release once, they will serialize. That's fine.
    // However, we need to make sure the program prints "DONE done=1" after both finish.
    
    let s_clone1 = Arc::clone(&s);
    let s_clone2 = Arc::clone(&s);

    let handle1 = thread::spawn(move || {
        // w1
        s_clone1.acquire();
        // Simulate work
        // Maybe do nothing or print something? No, final output is just DONE done=1
        s_clone1.release();
    });

    let handle2 = thread::spawn(move || {
        // w2
        s_clone2.acquire();
        // Simulate work
        s_clone2.release();
    });

    // Wait for both to finish
    handle1.join().unwrap();
    handle2.join().unwrap();

    println!("DONE done=1");
}
```
