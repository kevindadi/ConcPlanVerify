```rust
use std::sync::{Arc, Condvar, Mutex};

struct Semaphore {
    count: Mutex<usize>,
    cv: Condvar,
}

impl Semaphore {
    fn new(n: usize) -> Self {
        Semaphore { count: Mutex::new(n), cv: Condvar::new() }
    }

    fn acquire(&self) {
        let mut count = self.count.lock().unwrap();
        while *count == 0 {
            count = self.cv.wait(count).unwrap();
        }
        *count -= 1;
    }

    fn release(&self) {
        let mut count = self.count.lock().unwrap();
        *count += 1;
        self.cv.notify_one();
    }
}

fn worker(name: &'static str, s: Arc<Semaphore>) {
    let rounds: usize = 3;
    for _ in 0..rounds {
        s.acquire(); // hold permit while working (R3)
        // simulate work while holding the permit
        for _ in 0..5 {
            std::hint::spin_loop();
        }
        s.release(); // released on every path, exactly once per acquire (R4)
    }
    let _ = name;
}

fn main() {
    let s = Arc::new(Semaphore::new(1)); // one permit (R2)

    let s1 = Arc::clone(&s);
    let s2 = Arc::clone(&s);

    let h1 = std::thread::spawn(move || worker("w1", s1));
    let h2 = std::thread::spawn(move || worker("w2", s2));

    h1.join().unwrap();
    h2.join().unwrap();

    // R7: exactly `DONE done=1`
    let remaining = *s.count.lock().unwrap();
    println!("DONE done={}", remaining);
}
```

Notes on the requirements:
- R1: `main` starts the two worker threads and joins both via a supervising main scope.
- R2: `s` is a single shared counting permit pool initialized to exactly one permit.
- R3: each worker only performs its work while holding the permit acquired from `s`.
- R4: each worker acquires/releases the permit 3 times; each `acquire` is followed by exactly one `release` (including on the loop's normal exit path).
- R5: `acquire` uses a `Condvar` wait, so a waiting worker blocks without holding the mutex, letting the permit-holder release the permit so the waiter can proceed.
- R6: fair Mutex/Condvar handoff guarantees both workers drain their 3 rounds and terminate regardless of interleaving; no permit is lost, so `w1` and `w2` always finish.
- R7: the program prints exactly `DONE done=1` (the pool's final permit count) and exits.
