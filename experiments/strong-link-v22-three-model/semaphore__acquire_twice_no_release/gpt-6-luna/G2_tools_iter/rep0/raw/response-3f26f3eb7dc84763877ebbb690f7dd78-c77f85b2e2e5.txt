```rust
use std::hint::black_box;
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

struct State {
    permits: usize,
    next_ticket: u64,
    serving_ticket: u64,
}

struct Semaphore {
    state: Mutex<State>,
    changed: Condvar,
}

impl Semaphore {
    fn new(permits: usize) -> Self {
        Self {
            state: Mutex::new(State {
                permits,
                next_ticket: 0,
                serving_ticket: 0,
            }),
            changed: Condvar::new(),
        }
    }

    fn acquire(&self) -> Permit<'_> {
        let mut state = self.state.lock().unwrap();
        let ticket = state.next_ticket;
        state.next_ticket += 1;

        while state.permits == 0 || ticket != state.serving_ticket {
            state = self.changed.wait(state).unwrap();
        }

        state.permits -= 1;
        state.serving_ticket += 1;
        self.changed.notify_all();

        Permit { semaphore: self }
    }

    fn release(&self) {
        let mut state = self.state.lock().unwrap();
        state.permits += 1;
        self.changed.notify_all();
    }
}

struct Permit<'a> {
    semaphore: &'a Semaphore,
}

impl Drop for Permit<'_> {
    fn drop(&mut self) {
        self.semaphore.release();
    }
}

fn work(worker: u64, round: u64) {
    let mut value = worker.wrapping_add(round);
    for n in 0..10_000 {
        value = value.wrapping_mul(31).wrapping_add(n);
    }
    black_box(value);
}

fn w1(s: Arc<Semaphore>) {
    for round in 0..2 {
        let _permit = s.acquire();
        work(1, round);
    }
}

fn w2(s: Arc<Semaphore>) {
    for round in 0..2 {
        let _permit = s.acquire();
        work(2, round);
    }
}

fn main() {
    let s = Arc::new(Semaphore::new(1));

    let t1 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w1(s))
    };
    let t2 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w2(s))
    };

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
```
