use std::sync::{Arc, Condvar, Mutex};
use std::thread;

fn main() {

    let m = Arc::new(Mutex::new(false));
    let cv = Arc::new(Condvar::new());

        let cv = Arc::clone(&cv);

            let mut ready = m.lock().unwrap();

            while !*ready {

        let cv = Arc::clone(&cv);

            let mut ready = m.lock().unwrap();

            cv.notify_one();
