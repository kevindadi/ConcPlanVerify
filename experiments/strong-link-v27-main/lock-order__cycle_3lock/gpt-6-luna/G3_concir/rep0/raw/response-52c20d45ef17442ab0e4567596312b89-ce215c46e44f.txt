use std::sync::Mutex;
use std::thread;

fn t1(a: &Mutex<()>, b: &Mutex<()>) {
    let _a = a.lock().unwrap();
    let _b = b.lock().unwrap();
}

fn t2(b: &Mutex<()>, c: &Mutex<()>) {
    let _b = b.lock().unwrap();
    let _c = c.lock().unwrap();
}

fn t3(a: &Mutex<()>, c: &Mutex<()>) {
    let _a = a.lock().unwrap();
    let _c = c.lock().unwrap();
}

fn main() {
    let a = Mutex::new(());
    let b = Mutex::new(());
    let c = Mutex::new(());

    thread::scope(|scope| {
        let h1 = scope.spawn(|| t1(&a, &b));
        let h2 = scope.spawn(|| t2(&b, &c));
        let h3 = scope.spawn(|| t3(&a, &c));

        h1.join().unwrap();
        h2.join().unwrap();
        h3.join().unwrap();
    });

    println!("DONE done=1");
}
