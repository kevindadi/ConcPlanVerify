use std::sync::{Arc, Mutex};
use std::thread;

fn w1(m: Arc<Mutex<u8>>) {
    let mut c = m.lock().unwrap();
    assert!(*c < 2);
    *c += 1;
}

fn w2(m: Arc<Mutex<u8>>) {
    let mut c = m.lock().unwrap();
    assert!(*c < 2);
    *c += 1;
}

fn main() {
    let m = Arc::new(Mutex::new(0_u8));

    let m1 = Arc::clone(&m);
    let t1 = thread::spawn(move || w1(m1));

    let m2 = Arc::clone(&m);
    let t2 = thread::spawn(move || w2(m2));

    let r1 = t1.join();
    let r2 = t2.join();

    r1.unwrap();
    r2.unwrap();

    println!("DONE done=1");
}
