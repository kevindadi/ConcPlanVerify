```rust
use std::thread;

fn worker() {
    // worker performs no shared work; it just returns
}

fn main() {
    // First start-and-wait cycle
    let h1 = thread::spawn(|| {
        worker();
    });
    h1.join().expect("worker h1 panicked");

    // Second start-and-wait cycle
    let h2 = thread::spawn(|| {
        worker();
    });
    h2.join().expect("worker h2 panicked");

    println!("DONE done=1");
}
```
