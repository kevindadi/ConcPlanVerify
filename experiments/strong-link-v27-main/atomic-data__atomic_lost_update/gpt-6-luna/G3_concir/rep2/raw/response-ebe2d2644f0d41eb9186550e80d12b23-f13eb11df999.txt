use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicI32>) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let observed = match c.compare_exchange(
            old,
            old + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(value) | Err(value) => value,
        };
        if observed == old {
            return;
        }
    }
}

            Ordering::SeqCst,

}

fn main() {
