```rust
use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::Arc;
use std::thread;

fn increment(c: &AtomicUsize) {
    let mut current = c.load(Ordering::SeqCst);

    loop {
        match c.compare_exchange(
            current,
            current + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(_) => break,
            Err(actual) => current = actual,
        }
    }
}

fn w1(c: Arc<AtomicUsize>) {
    increment(&c);
}

fn w2(c: Arc<AtomicUsize>) {
    increment(&c);
}

fn main() {
    let c = Arc::new(AtomicUsize::new(0));

    let worker1 = thread::spawn({
        let c = Arc::clone(&c);
        move || w1(c)
    });
    let worker2 = thread::spawn({
        let c = Arc::clone(&c);
        move || w2(c)
    });

    worker1.join().unwrap();
    worker2.join().unwrap();

    assert_eq!(c.load(Ordering::SeqCst), 2);
    println!("DONE done=1");
}
```

The counter reaches two; `done=1` is printed literally as required. An absolute guarantee of R8 is not possible under an arbitrary scheduler that may indefinitely prevent a worker from running. With both workers getting scheduled to make progress, the strong compare-and-exchange loop terminates: there are only two successful increments and no spurious failures. If `done=1` is intended to report the counter’s value, R3 and R9 conflict.
