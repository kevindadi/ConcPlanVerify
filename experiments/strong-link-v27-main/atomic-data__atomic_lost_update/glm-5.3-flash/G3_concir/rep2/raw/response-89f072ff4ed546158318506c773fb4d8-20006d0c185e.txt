```rust
use std::sync::atomic::{AtomicI64, Ordering};
use std::sync::{Arc, Mutex};

// Resource: c (atomic counter, init 0), protected by resource: m (Mutex)
// per the ConcIR protection edge {var: c, lock: m}.

// CIR function: main::rmw_increment
//   mutex_lock m; write_shared c = c + 1; mutex_unlock m; return
// The write_shared is realized as an atomic read-modify-write retry loop:
// a failed update attempt is retried, never abandoned, and the update
// appears as a single indivisible step.
fn rmw_increment(m: &Mutex<()>, c: &AtomicI64) {
    let _guard = m.lock().unwrap();
    loop {
        let current = c.load(Ordering::SeqCst);
        match c.compare_exchange_weak(
            current,
            current + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(_) => break,
            Err(_) => continue, // failed attempt: retry
        }
    }
    // _guard dropped here: mutex_unlock m
}

// CIR function: main::w1
fn w1(m: Arc<Mutex<()>>, c: Arc<AtomicI64>) {
    rmw_increment(&m, &c);
}

// CIR function: main::w2
fn w2(m: Arc<Mutex<()>>, c: Arc<AtomicI64>) {
    rmw_increment(&m, &c);
}

// CIR entry: main::main
//   spawn w1 (h1); spawn w2 (h2); join h1; join h2; return
fn main() {
    let m = Arc::new(Mutex::new(()));
    let c = Arc::new(AtomicI64::new(0));

    let h1 = {
        let m = Arc::clone(&m);
        let c = Arc::clone(&c);
        std::thread::spawn(move || w1(m, c))
    };
    let h2 = {
        let m = Arc::clone(&m);
        let c = Arc::clone(&c);
        std::thread::spawn(move || w2(m, c))
    };

    h1.join().expect("w1 panicked");
    h2.join().expect("w2 panicked");

    // After joins: print the required terminal line.
    println!("DONE done=1");
}
```
