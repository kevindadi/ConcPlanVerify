```rust
use std::sync::mpsc::{sync_channel, Receiver, SyncSender};
use std::thread;

fn s1(tx: SyncSender<()>) {
    tx.send(()).unwrap();
}

fn r(rx: Receiver<()>) {
    let _value = rx.recv().unwrap();
}

fn main() {
    let ch = sync_channel::<()>(0);
    let (tx, rx) = ch;

    let s1_task = thread::spawn(move || s1(tx));
    let r_task = thread::spawn(move || r(rx));

    s1_task.join().unwrap();
    r_task.join().unwrap();

    println!("DONE done=1");
}
```
