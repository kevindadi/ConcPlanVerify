// Supervised workers sharing one counting semaphore (permit pool).
//
// R1: main acts as the supervisor: it launches w1 and w2 and joins both.
// R2: the semaphore `s` starts with exactly one permit.
// R3: a worker only performs its work while holding the permit.
// R4: each worker acquires the permit twice and releases it exactly
//     twice (release happens on every path: the loop has no early
//     return, panic, or break between acquire and release).
// R5: waiting is done on a Condvar while the internal Mutex guard is
//     parked, so the mutex is NOT held while waiting; the worker that
//     holds the permit can always reach `release()`.
// R6: every release increments the count before notifying, and each
//     worker performs only finitely many acquisitions, so no schedule
//     can deadlock or starve forever.
// R7: only the supervisor prints, producing exactly one line:
//     `DONE done=1` (the pool ends with 1 permit because every
//     acquire is matched by a release).

use std::sync::{Arc, Condvar, Mutex};
use std::thread;

struct Semaphore {
    permits: Mutex<usize>,
    cv: Condvar,
}

impl Semaphore {
    fn new(initial: usize) -> Self {
        Semaphore {
            permits: Mutex::new(initial),
            cv: Condvar::new(),
        }
    }

    fn acquire(&self) {
        let mut count = self.permits.lock().unwrap();
        while *count == 0 {
            // Waiting here releases the Mutex, so the current holder
            // of the permit is never blocked from calling release().
            count = self.cv.wait(count).unwrap();
        }
        *count -= 1;
    }

    fn release(&self) {
        let mut count = self.permits.lock().unwrap();
        *count += 1;
        drop(count);
        self.cv.notify_one();
    }

    fn available(&self) -> usize {
        *self.permits.lock().unwrap()
    }
}

fn worker(name: &'static str, s: Arc<Semaphore>) {
    // Each worker acquires the permit more than once (R4).
    for round in 0..2 {
        s.acquire();
        // Critical section: work is performed only while holding
        // the permit, so w1 and w2 never work simultaneously (R3).
        let _ = (name, round);
        // A couple of yields to encourage interleavings (R6 stress).
        thread::yield_now();
        thread::yield_now();
        s.release(); // released on every path through the loop body
    }
}

fn main() {
    // R2: exactly one permit initially.
    let s = Arc::new(Semaphore::new(1));

    let s1 = Arc::clone(&s);
    let s2 = Arc::clone(&s);

    // R1: supervisor launches both workers.
    let h1 = thread::spawn(move || worker("w1", s1));
    let h2 = thread::spawn(move || worker("w2", s2));

    // R1: supervisor waits for both workers to finish.
    h1.join().unwrap();
    h2.join().unwrap();

    // R7: exactly one output line. Every acquire was matched by a
    // release, so the pool is back to its initial size of 1.
    println!("DONE done={}", s.available());
}
