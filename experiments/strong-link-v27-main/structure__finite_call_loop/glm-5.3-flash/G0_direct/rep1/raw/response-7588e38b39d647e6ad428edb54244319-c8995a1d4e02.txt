// R1: The main task calls the auxiliary routine `helper`, then begins the
//     same call sequence again (two identical call rounds).
// R2: The main task and `helper` share no mutexes, counters, or other
//     shared state — no two tasks ever contend for a resource.
// R3: Each auxiliary call runs to completion (the thread is joined) before
//     the calling task starts the next call.
// R4: Every schedule and interleaving terminates: each spawned thread is
//     joined before proceeding, and the loop is bounded.
// R5: The program prints exactly the line `DONE done=1` and then exits.

fn helper() {
    // Auxiliary routine. It uses no shared state of any kind, so it can
    // never contend with the main task (or with anything else).
}

fn main() {
    // Two rounds of the same call sequence: call the auxiliary routine,
    // wait for it to run to completion, then begin the sequence again.
    for _round in 0..2 {
        let handle = std::thread::spawn(helper);
        // R3: run this call to completion before starting the next one.
        handle.join().expect("helper thread panicked");
    }

    // R5: exactly one line of output, then exit.
    println!("DONE done=1");
}
