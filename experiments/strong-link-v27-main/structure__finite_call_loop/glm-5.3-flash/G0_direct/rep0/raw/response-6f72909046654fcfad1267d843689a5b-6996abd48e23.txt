// R2: `helper` and the main task share no mutexes, counters, or any other

// R3: Each call to `helper` runs to completion before the main task starts

// R4: There is no concurrency, no loops, and no blocking, so every
//     schedule/interleaving trivially terminates.
// R5: The program prints exactly `DONE done=1` and then exits.

// Role: helper — the auxiliary routine.
// It takes no arguments, touches no shared state, and returns normally.
fn helper() {
    // Runs to completion on every call; no side effects, no shared state.
}

fn main() {
    // First call sequence: main task invokes the auxiliary routine.
    helper();

    // R1: the main task now begins the same call sequence again.
    helper();

    // R5: exactly one line of output, then the program exits.
    println!("DONE done=1");
}
