use std::sync::{Arc, Mutex};
use std::thread;

fn activation(s: Arc<Mutex<()>>) {
    let _permit = s.lock().unwrap_or_else(|e| e.into_inner());

    let work = (0..1_000u64).fold(0u64, |sum, n| sum.wrapping_add(n));
    std::hint::black_box(work);
}

fn run_role(s: Arc<Mutex<()>>) {
    let s1 = Arc::clone(&s);
    let s2 = Arc::clone(&s);

    let a1 = thread::spawn(move || activation(s1));
    let a2 = thread::spawn(move || activation(s2));

    a1.join().expect("activation panicked");
    a2.join().expect("activation panicked");
}

fn w1(s: Arc<Mutex<()>>) {
    run_role(s);
}

fn w2(s: Arc<Mutex<()>>) {
    run_role(s);
}

fn w3(s: Arc<Mutex<()>>) {
    run_role(s);
}

fn main() {
    let s = Arc::new(Mutex::new(()));

    let t1 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w1(s))
    };
    let t2 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w2(s))
    };
    let t3 = thread::spawn(move || w3(s));

    t1.join().expect("w1 panicked");
    t2.join().expect("w2 panicked");
    t3.join().expect("w3 panicked");

    println!("DONE done=1");
}
