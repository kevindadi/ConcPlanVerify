// R1: main task starts three worker roles (w1, w2, w3) sharing a single
//     permit (s); each role has up to two activations running at once.
// R2: each activation holds the permit while working, releases afterwards
//     (via Mutex guard drop).
// R3: at most one activation holds the permit (Mutex guarantees this).
// R4: an activation that cannot obtain the permit waits (Mutex::lock blocks).
// R5: no deadlock/cycle: permit is always released when the guard drops.
// R6: main joins all activations, so all three roles complete.
// R7: prints exactly `DONE done=1` and exits.

use std::sync::{Arc, Mutex};
use std::thread;

fn activation(s: Arc<Mutex<()>>) {
    // R4: blocks until the permit is available.
    let _permit = s.lock().unwrap();
    // R2: holding the permit while doing its work.
    // (work happens here; nothing to compute, just hold the permit)
    // R2: permit released when `_permit` is dropped at end of scope.
}

fn role(name: &'static str, s: Arc<Mutex<()>>) -> thread::JoinHandle<()> {
    // Each role runs two activations (R1: up to two at once per role).
    let s1 = Arc::clone(&s);
    let s2 = Arc::clone(&s);
    thread::spawn(move || {
        let h1 = thread::spawn(move || activation(s1));
        let h2 = thread::spawn(move || activation(s2));
        h1.join().unwrap();
        h2.join().unwrap();
        let _ = name; // role identity (w1/w2/w3)
    })
}

fn main() {
    // Shared resource: s, a single-permit semaphore (R1).
    let s = Arc::new(Mutex::new(()));

    let w1 = role("w1", Arc::clone(&s));
    let w2 = role("w2", Arc::clone(&s));
    let w3 = role("w3", Arc::clone(&s));

    // R5/R6: wait for all roles (and thus all activations) to finish.
    w1.join().unwrap();
    w2.join().unwrap();
    w3.join().unwrap();

    // R7: exactly this line, then exit.
    println!("DONE done=1");
}
