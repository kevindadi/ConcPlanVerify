use concir_sync::Semaphore;

fn main() {
    let s = Semaphore::new(1);

    let w1 = {
        let s = s.clone();
        move || {
            let mut work: i64 = 0;
            let p = s.acquire();
            work += 1;
            p.release();
            let p = s.acquire();
            work += 1;
            p.release();
            let _ = work;
        }
    };

    let w2 = {
        let s = s.clone();
        move || {
            let mut work: i64 = 0;
            let p = s.acquire();
            work += 1;
            p.release();
            let p = s.acquire();
            work += 1;
            p.release();
            let _ = work;
        }
    };

    let h1 = std::thread::spawn(w1);
    let h2 = std::thread::spawn(w2);
    h1.join().unwrap();
    h2.join().unwrap();
    println!("DONE done=1");
}
