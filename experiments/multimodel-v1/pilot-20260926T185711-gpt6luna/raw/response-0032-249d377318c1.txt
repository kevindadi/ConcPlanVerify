use std::sync::{Arc, Mutex};
use std::thread;

fn sender(ch: SyncSender<i32>, m: Arc<Mutex<()>>) {

        let _guard = m.lock().unwrap();
    }
    ch.send(1).unwrap();

    {
        let _guard = m.lock().unwrap();
    }
    ch.send(2).unwrap();
}

fn receiver(ch: Receiver<i32>, m: Arc<Mutex<()>>) {
    let mut value = 0;
    value = ch.recv().unwrap();
    {
        let _guard = m.lock().unwrap();
    }

    value = ch.recv().unwrap();
    {
        let _guard = m.lock().unwrap();
    }

}
