use concir_sync::Semaphore;
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

struct Cv {
    w1: Condvar,
    w2: Condvar,
    announced: Condvar,
}

struct Shared {
    m1: Mutex<bool>,
    m2: Mutex<bool>,
    cv: Cv,
    ready: Arc<Semaphore>,
    announced: Mutex<usize>,
}

fn w1(shared: Arc<Shared>) {
    let _permit = shared.ready.acquire();
    let mut guard = shared.m1.lock().unwrap();

    {
        let mut announced = shared.announced.lock().unwrap();
        *announced += 1;
        shared.cv.announced.notify_one();
    }

    while !*guard {
        guard = shared.cv.w1.wait(guard).unwrap();
    }
}
