```rust
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    // Shared locks a and b
    let a = Arc::new(Mutex::new(0));
    let b = Arc::new(Mutex::new(0));

    // Clone for t1
    let a1 = Arc::clone(&a);
    let b1 = Arc::clone(&b);

    // Clone for t2
    let a2 = Arc::clone(&a);
    let b2 = Arc::clone(&b);

    // Worker t1
    let h1 = thread::spawn(move || {
        // To prevent deadlock (R5), we must acquire locks in a consistent order.
        // Let's say the global order is: lock a, then lock b.
        
        // Acquire lock a
        let _guard_a = a1.lock().unwrap();
        
        // Acquire lock b while holding a
        let _guard_b = b1.lock().unwrap();
        
        // Critical work: both locks are held here
        
        // Locks are released automatically when guards go out of scope at end of block
    });

    // Worker t2
    let h2 = thread::spawn(move || {
        // Same consistent order: lock a, then lock b
        
        // Acquire lock a
        let _guard_a = a2.lock().unwrap();
        
        // Acquire lock b while holding a
        let _guard_b = b2.lock().unwrap();
        
        // Critical work: both locks are held here
        
        // Locks are released automatically when guards go out of scope at end of block
    });

    // Main thread waits for both workers to finish (R6)
    h1.join().unwrap();
    h2.join().unwrap();

    // Print exactly the required line (R9)
    println!("DONE t1=1 t2=1");
}
```
