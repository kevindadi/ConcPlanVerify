use std::sync::mpsc::{channel, Sender, Receiver};
use std::sync::Mutex;
use std::thread;

static LK: Mutex<()> = Mutex::new(());

fn send(tx: &Sender<i32>, v: i32) {
    let _ = tx.send(v);
}

fn recv(rx: &Receiver<i32>) -> i32 {
    rx.recv().unwrap()
}

fn s(tx1: Sender<i32>, rx2: Receiver<i32>) {
    {
        let _g = LK.lock().unwrap();
    }
    send(&tx1, 1);
    let _ = recv(&rx2);
}

fn r(rx1: Receiver<i32>, tx2: Sender<i32>) {
    {
        let _g = LK.lock().unwrap();
    }
    let _ = recv(&rx1);
    send(&tx2, 1);
}

fn main() {
    let (tx1, rx1) = channel::<i32>();
    let (tx2, rx2) = channel::<i32>();

    let hs = thread::spawn(move || s(tx1, rx2));
    let hr = thread::spawn(move || r(rx1, tx2));

    hs.join().unwrap();
    hr.join().unwrap();

    println!("DONE done=1");
}
