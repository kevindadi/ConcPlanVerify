use std::sync::mpsc::{channel, Receiver, Sender};
use std::sync::Mutex;
use std::thread;

static lk: Mutex<()> = Mutex::new(());

fn send(tx: &Sender<i32>, v: i32) {
    let _ = tx.send(v);
}

fn recv(rx: &Receiver<i32>) -> i32 {
    rx.recv().unwrap()
}

fn s(ch1_tx: Sender<i32>, ch2_rx: Receiver<i32>) {
    {
        let _g = lk.lock().unwrap();
    }
    send(&ch1_tx, 1);
    let _ = recv(&ch2_rx);
}

fn r(ch1_rx: Receiver<i32>, ch2_tx: Sender<i32>) {
    {
        let _g = lk.lock().unwrap();
    }
    let _ = recv(&ch1_rx);
    send(&ch2_tx, 1);
}

fn main() {
    let (ch1_tx, ch1_rx) = channel::<i32>();
    let (ch2_tx, ch2_rx) = channel::<i32>();

    let hs = thread::spawn(move || s(ch1_tx, ch2_rx));
    let hr = thread::spawn(move || r(ch1_rx, ch2_tx));

    hs.join().unwrap();
    hr.join().unwrap();

    println!("DONE done=1");
}
