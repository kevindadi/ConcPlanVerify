```rust
use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::{Arc, Mutex};

fn w1(m: &Arc<Mutex<()>>, c: &Arc<AtomicI32>) {
    let _guard = m.lock().unwrap();
    let cur = c.load(Ordering::SeqCst);
    if cur < 1 {
        c.store(cur + 1, Ordering::SeqCst);
    }
}

fn w2(m: &Arc<Mutex<()>>, c: &Arc<AtomicI32>) {
    let _guard = m.lock().unwrap();
    let cur = c.load(Ordering::SeqCst);
    if cur < 1 {
        c.store(cur + 1, Ordering::SeqCst);
    }
}

fn main() {
    let m = Arc::new(Mutex::new(()));
    let c = Arc::new(AtomicI32::new(0));

    let m1 = Arc::clone(&m);
    let c1 = Arc::clone(&c);
    let h1 = std::thread::spawn(move || w1(&m1, &c1));

    let m2 = Arc::clone(&m);
    let c2 = Arc::clone(&c);
    let h2 = std::thread::spawn(move || w2(&m2, &c2));

    h1.join().unwrap();
    h2.join().unwrap();

    let done = c.load(Ordering::SeqCst);
    println!("DONE done={}", done);
}
```
