```rust
use concir_sync::Semaphore;
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

struct Shared {
    proceed: bool,
    ready: usize,
}

fn w1(m: Arc<Mutex<Shared>>, cv: Arc<Condvar>, g12: Arc<Semaphore>, g_n: Arc<Semaphore>) {
    let _ = (g12, g_n);
    let mut state = m.lock().unwrap();
    state.ready += 1;
    cv.notify_all();
    while !state.proceed {
        state = cv.wait(state).unwrap();
    }
}

fn w2(m: Arc<Mutex<Shared>>, cv: Arc<Condvar>, g12: Arc<Semaphore>, g_n: Arc<Semaphore>) {
    let _ = (g12, g_n);
    let mut state = m.lock().unwrap();
    state.ready += 1;
    cv.notify_all();
    while !state.proceed {
        state = cv.wait(state).unwrap();
    }
}

fn notifier(m: Arc<Mutex<Shared>>, cv: Arc<Condvar>, g12: Arc<Semaphore>, g_n: Arc<Semaphore>) {
    let _ = (g12, g_n);
    let mut state = m.lock().unwrap();
    while state.ready < 2 {
        state = cv.wait(state).unwrap();
    }
    state.proceed = true;
    cv.notify_all();
}

fn main() {
    let m = Arc::new(Mutex::new(Shared {
        proceed: false,
        ready: 0,
    }));
    let cv = Arc::new(Condvar::new());
    let g12 = Semaphore::new(0);
    let g_n = Semaphore::new(0);

    let w1_thread = {
        let m = Arc::clone(&m);
        let cv = Arc::clone(&cv);
        let g12 = Arc::clone(&g12);
        let g_n = Arc::clone(&g_n);
        thread::spawn(move || w1(m, cv, g12, g_n))
    };
    let w2_thread = {
        let m = Arc::clone(&m);
        let cv = Arc::clone(&cv);
        let g12 = Arc::clone(&g12);
        let g_n = Arc::clone(&g_n);
        thread::spawn(move || w2(m, cv, g12, g_n))
    };
    let notifier_thread = {
        let m = Arc::clone(&m);
        let cv = Arc::clone(&cv);
        let g12 = Arc::clone(&g12);
        let g_n = Arc::clone(&g_n);
        thread::spawn(move || notifier(m, cv, g12, g_n))
    };

    w1_thread.join().unwrap();
    w2_thread.join().unwrap();
    notifier_thread.join().unwrap();

    println!("DONE waiters=0");
}
```
