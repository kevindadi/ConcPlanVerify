use std::thread;

fn x1(a: &Arc<Mutex<()>>, b: &Arc<Mutex<()>>) {
    let _ga = a.lock().unwrap();
    let _gb = b.lock().unwrap();
    drop(_gb);
    drop(_ga);
}

fn x2(a: &Arc<Mutex<()>>, b: &Arc<Mutex<()>>) {
    let _ga = a.lock().unwrap();
    let _gb = b.lock().unwrap();
    drop(_gb);
    drop(_ga);
}

fn outer(a: Arc<Mutex<()>>, b: Arc<Mutex<()>>) {
    let h1 = thread::spawn(move || {
        let a = a.clone();
        let b = b.clone();
        x1(&a, &b)

        let a = a.clone();
        let b = b.clone();
        x2(&a, &b)
    });

    h2.join().unwrap();
}
