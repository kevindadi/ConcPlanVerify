use std::thread;

fn w1(m: &Arc<Mutex<i32>>) {
    // s1: mutex_lock m
    let mut guard = m.lock().unwrap();
    // s2: branch c < 1
    if *guard < 1 {
        // s3: write_shared c = c + 1
        *guard = *guard + 1;
    }
    // s4: mutex_unlock m (guard drops)
    drop(guard);
    // s5: return
}

fn w2(m: &Arc<Mutex<i32>>) {
    // s1: mutex_lock m
    let mut guard = m.lock().unwrap();
    // s2: branch c < 1
    if *guard < 1 {
        // s3: write_shared c = c + 1
        *guard = *guard + 1;
    }
    // s4: mutex_unlock m (guard drops)
    drop(guard);

}

fn main() {
    // shared resource c (init 0) protected by mutex m
    let m: Arc<Mutex<i32>> = Arc::new(Mutex::new(0));

    let h1 = thread::spawn(move || w1(&m1));

    let m2 = Arc::clone(&m);
