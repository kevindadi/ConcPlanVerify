use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicI32>) {
    let mut old = 0;
    loop {
        let seen = match c.compare_exchange(old, old + 1, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(previous) => previous,
            Err(current) => current,
        };
        if seen == old {
            return;
        }
        old = seen;
    }
}

fn w2(c: Arc<AtomicI32>) {
    let mut old = 0;
    loop {
        let seen = match c.compare_exchange(old, old + 1, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(previous) => previous,
            Err(current) => current,
        };
        if seen == old {
            return;
        }
        old = seen;
    }
}

fn main() {
    let c = Arc::new(AtomicI32::new(0));
    let w1_c = Arc::clone(&c);
    let w2_c = Arc::clone(&c);

    let t1 = thread::spawn(move || w1(w1_c));
    let t2 = thread::spawn(move || w2(w2_c));

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
