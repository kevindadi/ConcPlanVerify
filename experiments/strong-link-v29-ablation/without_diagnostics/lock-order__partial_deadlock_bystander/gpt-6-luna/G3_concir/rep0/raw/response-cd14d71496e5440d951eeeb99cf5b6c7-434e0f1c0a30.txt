use std::sync::{Arc, Mutex};
use std::thread;

    flag: i32,
}

    b: Arc<Mutex<()>>,
    sa: Arc<Semaphore>,
    sb: Arc<Semaphore>,
    sa_permit: concir_sync::Permit,
) {
    let a_guard = a.lock().unwrap();
    sa_permit.release();
    drop(a_guard);

}
