use std::sync::mpsc::{sync_channel, Receiver, SyncSender};
use std::sync::{Arc, Mutex};
use std::thread;

fn s(m: Arc<Mutex<()>>, ch1_tx: SyncSender<i32>, ch2_rx: Receiver<i32>) {
    let guard = m.lock().unwrap();
    drop(guard);

    ch1_tx.send(1).unwrap();
    let _value = ch2_rx.recv().unwrap();
}

fn r(m: Arc<Mutex<()>>, ch1_rx: Receiver<i32>, ch2_tx: SyncSender<i32>) {
    let guard = m.lock().unwrap();
    drop(guard);

    let _value = ch1_rx.recv().unwrap();
    ch2_tx.send(2).unwrap();
}

fn main() {
    let m = Arc::new(Mutex::new(()));
    let (ch1_tx, ch1_rx) = sync_channel(0);
    let (ch2_tx, ch2_rx) = sync_channel(0);

    let s_handle = {
        let m = Arc::clone(&m);
        thread::spawn(move || s(m, ch1_tx, ch2_rx))
    };
    let r_handle = {
        let m = Arc::clone(&m);
        thread::spawn(move || r(m, ch1_rx, ch2_tx))
    };

    s_handle.join().unwrap();
    r_handle.join().unwrap();

    println!("DONE done=1");
}
