use std::sync::{Arc, Mutex};
use std::thread;

fn w1(m: Arc<Mutex<i32>>) {
    let mut guard = m.lock().unwrap();
    let c = *guard;
    *guard = c + 1;
}

fn w2(m: Arc<Mutex<i32>>) {
    let mut guard = m.lock().unwrap();
    let c = *guard;
    *guard = c + 1;
}

fn main() {
    let m = Arc::new(Mutex::new(0));

    let w1_m = Arc::clone(&m);
    let t1 = thread::spawn(move || w1(w1_m));

    let w2_m = Arc::clone(&m);
    let t2 = thread::spawn(move || w2(w2_m));

    t1.join().unwrap();
    t2.join().unwrap();

    println!("DONE done=1");
}
