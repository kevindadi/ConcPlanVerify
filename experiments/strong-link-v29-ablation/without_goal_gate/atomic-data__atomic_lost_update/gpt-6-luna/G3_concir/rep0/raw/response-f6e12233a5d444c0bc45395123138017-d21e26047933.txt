use std::sync::atomic::{AtomicI32, Ordering};
use std::sync::Arc;
use std::thread;

fn w1(c: Arc<AtomicI32>) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let seen = match c.compare_exchange(
            old,
            old + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(previous) => previous,
            Err(actual) => actual,
        };

        if seen == old {
            return;
        }
    }
}

fn w2(c: Arc<AtomicI32>) {
    loop {
        let old = c.load(Ordering::SeqCst);
        let seen = match c.compare_exchange(
            old,
            old + 1,
            Ordering::SeqCst,
            Ordering::SeqCst,
        ) {
            Ok(previous) => previous,
            Err(actual) => actual,
        };

        if seen == old {
            return;
        }
    }
}

fn main() {
    let c = Arc::new(AtomicI32::new(0));

    let c1 = Arc::clone(&c);
    let c2 = Arc::clone(&c);

    let handle1 = thread::spawn(move || w1(c1));
    let handle2 = thread::spawn(move || w2(c2));

    handle1.join().unwrap();
    handle2.join().unwrap();

    println!("DONE done=1");
}
