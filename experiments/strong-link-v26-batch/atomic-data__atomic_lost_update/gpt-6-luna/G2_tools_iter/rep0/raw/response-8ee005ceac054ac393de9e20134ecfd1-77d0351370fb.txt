```rust
use std::sync::atomic::{AtomicUsize, Ordering};
use std::sync::Arc;
use std::thread;

fn increment(c: &AtomicUsize) {
    let mut current = c.load(Ordering::SeqCst);

    loop {
        match c.compare_exchange(current, current + 1, Ordering::SeqCst, Ordering::SeqCst) {
            Ok(_) => break,
            Err(actual) => current = actual,
        }
    }
}

fn w1(c: Arc<AtomicUsize>) {
    increment(&c);
}

fn w2(c: Arc<AtomicUsize>) {
    increment(&c);
}

fn main() {
    let c = Arc::new(AtomicUsize::new(0));

    let w1 = {
        let c = Arc::clone(&c);
        thread::Builder::new()
            .name("w1".into())
            .spawn(move || w1(c))
            .expect("failed to spawn w1")
    };

    let w2 = {
        let c = Arc::clone(&c);
        thread::Builder::new()
            .name("w2".into())
            .spawn(move || w2(c))
            .expect("failed to spawn w2")
    };

    w1.join().expect("w1 panicked");
    w2.join().expect("w2 panicked");

    assert_eq!(c.load(Ordering::SeqCst), 2);
    println!("DONE done=1");
}
```
