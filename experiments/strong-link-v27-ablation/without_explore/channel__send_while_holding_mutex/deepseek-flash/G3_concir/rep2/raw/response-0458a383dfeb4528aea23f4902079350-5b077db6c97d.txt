use std::sync::mpsc::{sync_channel, Receiver, SyncSender};
use std::sync::{Arc, Mutex};
use std::thread;

fn s(lk: Arc<Mutex<()>>, tx1: SyncSender<i32>, rx2: Receiver<i32>) {
    {
        let _guard = lk.lock().unwrap();
    }

    tx1.send(1).unwrap();

    let ack: i32 = rx2.recv().unwrap();
    let _ = ack;

    {
        let _guard = lk.lock().unwrap();
    }
}

fn r(lk: Arc<Mutex<()>>, rx1: Receiver<i32>, tx2: SyncSender<i32>) {
    {
        let _guard = lk.lock().unwrap();
    }

    let v: i32 = rx1.recv().unwrap();
    let _ = v;

    tx2.send(1).unwrap();

    {
        let _guard = lk.lock().unwrap();
    }
}

fn main() {
    let lk = Arc::new(Mutex::new(()));

    let (tx1, rx1) = sync_channel::<i32>(0);
    let (tx2, rx2) = sync_channel::<i32>(0);

    let lk_s = Arc::clone(&lk);
    let lk_r = Arc::clone(&lk);

    let hs = thread::spawn(move || s(lk_s, tx1, rx2));
    let hr = thread::spawn(move || r(lk_r, rx1, tx2));

    hs.join().unwrap();
    hr.join().unwrap();

    println!("DONE done=1");
}
