```rust
use std::sync::{Arc, Mutex, mpsc::{sync_channel, SyncSender, Receiver}};
use std::thread;

struct Shared {
    done: i32,
}

fn sender(tx: SyncSender<i32>, m: Arc<Mutex<Shared>>) {
    let mut i = 0;
    tx.send(1).unwrap();
    {
        let _guard = m.lock().unwrap();
        i += 1;
    }
    tx.send(2).unwrap();
    let _ = i;
}

fn receiver(rx: Receiver<i32>, m: Arc<Mutex<Shared>>) {
    let mut j = 0;
    {
        let _guard = m.lock().unwrap();
        j += 1;
    }
    let x = rx.recv().unwrap();
    {
        let _guard = m.lock().unwrap();
        j += 1;
    }
    let y = rx.recv().unwrap();
    let _ = (j, x, y);
}

fn main() {
    let m = Arc::new(Mutex::new(Shared { done: 0 }));
    let (tx, rx) = sync_channel::<i32>(1);

    let m_sender = Arc::clone(&m);
    let m_receiver = Arc::clone(&m);

    let hs = thread::spawn(move || sender(tx, m_sender));
    let hr = thread::spawn(move || receiver(rx, m_receiver));

    hs.join().unwrap();
    hr.join().unwrap();

    {
        let mut guard = m.lock().unwrap();
        guard.done = 1;
    }

    println!("DONE done=1");
}
```
