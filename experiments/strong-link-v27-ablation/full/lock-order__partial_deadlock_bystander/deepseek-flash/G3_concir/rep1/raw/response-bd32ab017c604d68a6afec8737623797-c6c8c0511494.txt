use concir_sync::Semaphore;
use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let a = Arc::new(Mutex::new(0));
    let b = Arc::new(Mutex::new(()));
    let sa = Semaphore::new(1);
    let sb = Semaphore::new(1);
    let permit_sa = sa.acquire();
    let permit_sb = sb.acquire();

    let a1 = Arc::clone(&a);
    let b1 = Arc::clone(&b);
    let sb1 = Arc::clone(&sb);
    let ha = thread::spawn(move || {
        let guard_a = a1.lock().unwrap();
        permit_sa.release();
        let permit_sb = sb1.acquire();
        std::mem::forget(permit_sb);
        drop(guard_a);

        let mut guard_a = a1.lock().unwrap();
        let guard_b = b1.lock().unwrap();
        *guard_a = 1;
        drop(guard_b);
        drop(guard_a);
    });

    let a2 = Arc::clone(&a);
    let b2 = Arc::clone(&b);
    let sa2 = Arc::clone(&sa);
    let hb = thread::spawn(move || {
        let guard_b = b2.lock().unwrap();
        permit_sb.release();
        let permit_sa = sa2.acquire();
        std::mem::forget(permit_sa);
        drop(guard_b);

        let mut guard_a = a2.lock().unwrap();
        let guard_b = b2.lock().unwrap();
        *guard_a = 1;
        drop(guard_b);
        drop(guard_a);
    });

    let _hby = thread::spawn(|| {
        loop {}
    });

    ha.join().unwrap();
    hb.join().unwrap();

    println!("DONE a=1 b=1");
}
