use std::sync::{Arc, Mutex, Condvar};
use std::thread;
use concir_sync::Semaphore;

fn main() {
    let m = Arc::new(Mutex::new(false));
    let cv = Arc::new(Condvar::new());
    let g12 = Semaphore::new(0);
    let gN = Semaphore::new(0);

    let m1 = Arc::clone(&m);
    let cv1 = Arc::clone(&cv);
    let g12_1 = Arc::clone(&g12);
    let gN1 = Arc::clone(&gN);
    let w1 = thread::spawn(move || {
        let mut guard = m1.lock().unwrap();
        gN1.release();
        while !*guard {
            guard = cv1.wait(guard).unwrap();
        }
        drop(guard);
        let permit = g12_1.acquire();
        std::mem::forget(permit);
    });

    let m2 = Arc::clone(&m);
    let cv2 = Arc::clone(&cv);
    let g12_2 = Arc::clone(&g12);
    let gN2 = Arc::clone(&gN);
    let w2 = thread::spawn(move || {
        let mut guard = m2.lock().unwrap();
        gN2.release();
        while !*guard {
            guard = cv2.wait(guard).unwrap();
        }
        drop(guard);
        let permit = g12_2.acquire();
        std::mem::forget(permit);
    });

    let m3 = Arc::clone(&m);
    let cv3 = Arc::clone(&cv);
    let g12_3 = Arc::clone(&g12);
    let gN3 = Arc::clone(&gN);
    let notifier = thread::spawn(move || {
        let p1 = gN3.acquire();
        let p2 = gN3.acquire();
        std::mem::forget(p1);
        std::mem::forget(p2);
        let mut guard = m3.lock().unwrap();
        *guard = true;
        cv3.notify_all();
        drop(guard);
        g12_3.release();
        g12_3.release();
    });

    w1.join().unwrap();
    w2.join().unwrap();
    notifier.join().unwrap();

    println!("DONE waiters=0");
}
