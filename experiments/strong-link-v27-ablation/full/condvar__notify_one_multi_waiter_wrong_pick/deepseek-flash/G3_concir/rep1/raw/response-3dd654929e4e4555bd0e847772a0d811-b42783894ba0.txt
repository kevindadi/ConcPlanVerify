```rust
use concir_sync::Semaphore;
use std::sync::{Arc, Condvar, Mutex};
use std::thread;

fn main() {
    // ---- shared resources -------------------------------------------------
    // `m` guards the condition flag, `cv` broadcasts it.
    let m = Arc::new(Mutex::new(false));
    let cv = Arc::new(Condvar::new());

    // `g12` and `gN` are permit based.  The design gives them count 0, which
    // means the permits they will hand out are already in flight: main creates
    // each semaphore with the two permits it needs and immediately takes both,
    // handing one to each waiter.  That leaves each semaphore with the
    // design's available count of 0, and a waiter can post its permit exactly
    // at its `semaphore_release` point.
    //
    // The semaphores are leaked so that the permits they hand out carry a
    // `'static` borrow and may be moved into the spawned threads.
    let g12: &'static Arc<Semaphore> = Box::leak(Box::new(Semaphore::new(2)));
    let gN: &'static Arc<Semaphore> = Box::leak(Box::new(Semaphore::new(2)));

    let p12_w1 = g12.acquire();
    let p12_w2 = g12.acquire();
    let pN_w1 = gN.acquire();
    let pN_w2 = gN.acquire();

    // ---- w1 ---------------------------------------------------------------
    let m1 = Arc::clone(&m);
    let cv1 = Arc::clone(&cv);
    let w1 = thread::spawn(move || {
        let mut guard = m1.lock().unwrap(); // mutex_lock   m
        p12_w1.release();                   // semaphore_release g12
        while !*guard {
            guard = cv1.wait(guard).unwrap(); // condvar_wait cv (holds m)
        }
        drop(guard);                        // mutex_unlock m
        pN_w1.release();                    // semaphore_release gN
    });

    // ---- w2 ---------------------------------------------------------------
    let m2 = Arc::clone(&m);
    let cv2 = Arc::clone(&cv);
    let w2 = thread::spawn(move || {
        let mut guard = m2.lock().unwrap(); // mutex_lock   m
        p12_w2.release();                   // semaphore_release g12
        while !*guard {
            guard = cv2.wait(guard).unwrap(); // condvar_wait cv (holds m)
        }
        drop(guard);                        // mutex_unlock m
        pN_w2.release();                    // semaphore_release gN
    });

    // ---- notifier ---------------------------------------------------------
    let m3 = Arc::clone(&m);
    let cv3 = Arc::clone(&cv);
    let g12_n = Arc::clone(g12);
    let gN_n = Arc::clone(gN);
    let notifier = thread::spawn(move || {
        let _a1 = g12_n.acquire();          // semaphore_acquire g12
        let _a2 = g12_n.acquire();          // semaphore_acquire g12
        let mut guard = m3.lock().unwrap(); // mutex_lock m
        *guard = true;                      // condition satisfied (under m)
        cv3.notify_all();                   // condvar_notify_all cv
        drop(guard);                        // mutex_unlock m
        let _b1 = gN_n.acquire();           // semaphore_acquire gN
        let _b2 = gN_n.acquire();           // semaphore_acquire gN
    });

    w1.join().unwrap();
    w2.join().unwrap();
    notifier.join().unwrap();

    println!("DONE waiters=0");
}
```
