use std::thread;

#[allow(non_upper_case_globals)]
pub mod main {
    use std::sync::Mutex;

    pub static a: Mutex<()> = Mutex::new(());

    pub fn t1() {
        // Declare dependency on the other module's resource.
        let guard_a = a.lock().unwrap();
        let guard_b = crate::other::b.lock().unwrap();

        let mut work: i32 = 0;
        work = 1;
        let _ = work;

        // Release each resource before finishing.
        drop(guard_b);
        drop(guard_a);
    }
}

#[allow(non_upper_case_globals)]
pub mod other {
    use std::sync::Mutex;

    pub static b: Mutex<()> = Mutex::new(());

    pub fn t2() {
        // Declare dependency on the other module's resource.
        let guard_a = crate::main::a.lock().unwrap();
        let guard_b = b.lock().unwrap();

        let mut work: i32 = 0;
        work = 1;
        let _ = work;

        // Release each resource before finishing.
        drop(guard_b);
        drop(guard_a);
    }
}

fn main() {
    let h1 = thread::spawn(crate::main::t1);
    let h2 = thread::spawn(crate::other::t2);

    h1.join().unwrap();
    h2.join().unwrap();

    println!("DONE done=1");
}
