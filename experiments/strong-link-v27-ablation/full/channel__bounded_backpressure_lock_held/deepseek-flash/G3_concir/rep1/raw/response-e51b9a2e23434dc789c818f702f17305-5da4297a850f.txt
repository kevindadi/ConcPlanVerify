use std::thread;

fn sender(m: Arc<Mutex<()>>, ch_tx: std::sync::mpsc::SyncSender<i32>) {
    {
        let _guard = m.lock().unwrap();
    }
    ch_tx.send(1).unwrap();
    ch_tx.send(2).unwrap();
}

fn receiver(m: Arc<Mutex<()>>, ch_rx: std::sync::mpsc::Receiver<i32>) {
    {
        let _guard = m.lock().unwrap();
    }
    let v1 = ch_rx.recv().unwrap();
    let v2 = ch_rx.recv().unwrap();
    let _ = (v1, v2);
}
