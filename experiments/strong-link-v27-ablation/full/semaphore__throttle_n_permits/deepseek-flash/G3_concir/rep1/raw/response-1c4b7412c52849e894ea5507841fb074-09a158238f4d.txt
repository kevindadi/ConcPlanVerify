```rust
use concir_sync::Semaphore;
use std::sync::Arc;
use std::thread;

fn w1(s: Arc<Semaphore>) {
    let permit = s.acquire();
    let mut work = 0;
    work = 1;
    permit.release();
    let _ = work;
}

fn w2(s: Arc<Semaphore>) {
    let permit = s.acquire();
    let mut work = 0;
    work = 1;
    permit.release();
    let _ = work;
}

fn w3(s: Arc<Semaphore>) {
    let permit = s.acquire();
    let mut work = 0;
    work = 1;
    permit.release();
    let _ = work;
}

fn main() {
    let s = Semaphore::new(2);

    let h1 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w1(s))
    };

    let h2 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w2(s))
    };

    let h3 = {
        let s = Arc::clone(&s);
        thread::spawn(move || w3(s))
    };

    h1.join().unwrap();
    h2.join().unwrap();
    h3.join().unwrap();

    println!("DONE done=1");
}
```
